📋 Required SQL Schema — Run in Supabase SQL Editor 
-- Noor Hospital CMS — Supabase Tables Setup
-- Run this entire script in your Supabase SQL Editor.

CREATE TABLE IF NOT EXISTS settings (
  key TEXT PRIMARY KEY,
  value TEXT,
  updated_at TIMESTAMPTZ DEFAULT now(),
  device_id TEXT
);

CREATE TABLE IF NOT EXISTS advance_cash (
  id BIGSERIAL PRIMARY KEY,
  date TEXT,
  amount NUMERIC DEFAULT 0,
  remarks TEXT,
  created_at TIMESTAMPTZ DEFAULT now(),
  updated_at TIMESTAMPTZ DEFAULT now(),
  device_id TEXT
);

CREATE TABLE IF NOT EXISTS hospital_cash (
  id BIGSERIAL PRIMARY KEY,
  date TEXT,
  source TEXT,
  amount NUMERIC DEFAULT 0,
  remarks TEXT,
  created_at TIMESTAMPTZ DEFAULT now(),
  updated_at TIMESTAMPTZ DEFAULT now(),
  device_id TEXT
);

CREATE TABLE IF NOT EXISTS hospital_deposits (
  id BIGSERIAL PRIMARY KEY,
  date TEXT,
  amount NUMERIC DEFAULT 0,
  "receiptNumber" TEXT,
  remarks TEXT,
  "attachmentUrl" TEXT,
  "fileName" TEXT,
  "fileType" TEXT,
  "uploadDate" TEXT,
  "pendingUpload" BOOLEAN DEFAULT false,
  "localAttachmentData" TEXT,
  created_at TIMESTAMPTZ DEFAULT now(),
  updated_at TIMESTAMPTZ DEFAULT now(),
  device_id TEXT
);

CREATE TABLE IF NOT EXISTS temporary_slips (
  id BIGSERIAL PRIMARY KEY,
  date TEXT,
  vendor TEXT,
  amount NUMERIC DEFAULT 0,
  "expenseType" TEXT,
  "tokenNumber" TEXT,
  remarks TEXT,
  status TEXT DEFAULT 'pending',
  "attachmentUrl" TEXT,
  "fileName" TEXT,
  "fileType" TEXT,
  "uploadDate" TEXT,
  "pendingUpload" BOOLEAN DEFAULT false,
  "localAttachmentData" TEXT,
  created_at TIMESTAMPTZ DEFAULT now(),
  updated_at TIMESTAMPTZ DEFAULT now(),
  device_id TEXT
);

CREATE TABLE IF NOT EXISTS bills (
  id BIGSERIAL PRIMARY KEY,
  date TEXT,
  "billNumber" TEXT,
  vendor TEXT,
  amount NUMERIC DEFAULT 0,
  "expenseType" TEXT,
  "tokenNumber" TEXT,
  category TEXT,
  remarks TEXT,
  "slipId" BIGINT,
  status TEXT DEFAULT 'pending',
  "transferId" BIGINT,
  "attachmentUrl" TEXT,
  "fileName" TEXT,
  "fileType" TEXT,
  "uploadDate" TEXT,
  "pendingUpload" BOOLEAN DEFAULT false,
  "localAttachmentData" TEXT,
  created_at TIMESTAMPTZ DEFAULT now(),
  updated_at TIMESTAMPTZ DEFAULT now(),
  device_id TEXT
);

-- Quick Upgrade for existing Supabase projects:
-- ALTER TABLE temporary_slips ADD COLUMN IF NOT EXISTS "tokenNumber" TEXT;
-- ALTER TABLE bills ADD COLUMN IF NOT EXISTS "tokenNumber" TEXT;

CREATE TABLE IF NOT EXISTS transfers (
  id BIGSERIAL PRIMARY KEY,
  date TEXT,
  type TEXT,
  amount NUMERIC DEFAULT 0,
  remarks TEXT,
  created_at TIMESTAMPTZ DEFAULT now(),
  updated_at TIMESTAMPTZ DEFAULT now(),
  device_id TEXT
);

-- Enable Row Level Security (optional but recommended)
ALTER TABLE settings ENABLE ROW LEVEL SECURITY;
ALTER TABLE advance_cash ENABLE ROW LEVEL SECURITY;
ALTER TABLE hospital_cash ENABLE ROW LEVEL SECURITY;
ALTER TABLE temporary_slips ENABLE ROW LEVEL SECURITY;
ALTER TABLE bills ENABLE ROW LEVEL SECURITY;
ALTER TABLE transfers ENABLE ROW LEVEL SECURITY;
ALTER TABLE hospital_deposits ENABLE ROW LEVEL SECURITY;

-- Allow all operations (for service_role key usage)
CREATE POLICY "Allow all" ON settings FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "Allow all" ON advance_cash FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "Allow all" ON hospital_cash FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "Allow all" ON temporary_slips FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "Allow all" ON bills FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "Allow all" ON transfers FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "Allow all" ON hospital_deposits FOR ALL USING (true) WITH CHECK (true);
