📋 Required SQL Schema — Run in Supabase SQL Editor 
-- Noor Hospital CMS — Supabase Tables Setup
-- Run this entire script in your Supabase SQL Editor.

CREATE TABLE IF NOT EXISTS settings (
  key TEXT PRIMARY KEY,
  value TEXT,
  updated_at TIMESTAMPTZ DEFAULT now(),
  device_id TEXT
);

CREATE TABLE IF NOT EXISTS advance_cash (
  id BIGSERIAL PRIMARY KEY,
  date TEXT,
  amount NUMERIC DEFAULT 0,
  remarks TEXT,
  created_at TIMESTAMPTZ DEFAULT now(),
  updated_at TIMESTAMPTZ DEFAULT now(),
  device_id TEXT
);

CREATE TABLE IF NOT EXISTS hospital_cash (
  id BIGSERIAL PRIMARY KEY,
  date TEXT,
  source TEXT,
  amount NUMERIC DEFAULT 0,
  remarks TEXT,
  created_at TIMESTAMPTZ DEFAULT now(),
  updated_at TIMESTAMPTZ DEFAULT now(),
  device_id TEXT
);

CREATE TABLE IF NOT EXISTS hospital_deposits (
  id BIGSERIAL PRIMARY KEY,
  date TEXT,
  amount NUMERIC DEFAULT 0,
  "receiptNumber" TEXT,
  remarks TEXT,
  "attachmentUrl" TEXT,
  "fileName" TEXT,
  "fileType" TEXT,
  "uploadDate" TEXT,
  "pendingUpload" BOOLEAN DEFAULT false,
  "localAttachmentData" TEXT,
  created_at TIMESTAMPTZ DEFAULT now(),
  updated_at TIMESTAMPTZ DEFAULT now(),
  device_id TEXT
);

CREATE TABLE IF NOT EXISTS temporary_slips (
  id BIGSERIAL PRIMARY KEY,
  date TEXT,
  vendor TEXT,
  amount NUMERIC DEFAULT 0,
  "expenseType" TEXT,
  "tokenNumber" TEXT,
  remarks TEXT,
  status TEXT DEFAULT 'pending',
  "attachmentUrl" TEXT,
  "fileName" TEXT,
  "fileType" TEXT,
  "uploadDate" TEXT,
  "pendingUpload" BOOLEAN DEFAULT false,
  "localAttachmentData" TEXT,
  created_at TIMESTAMPTZ DEFAULT now(),
  updated_at TIMESTAMPTZ DEFAULT now(),
  device_id TEXT
);

CREATE TABLE IF NOT EXISTS bills (
  id BIGSERIAL PRIMARY KEY,
  date TEXT,
  "billNumber" TEXT,
  vendor TEXT,
  amount NUMERIC DEFAULT 0,
  "expenseType" TEXT,
  "tokenNumber" TEXT,
  category TEXT,
  remarks TEXT,
  "slipId" BIGINT,
  status TEXT DEFAULT 'pending',
  "transferId" BIGINT,
  "attachmentUrl" TEXT,
  "fileName" TEXT,
  "fileType" TEXT,
  "uploadDate" TEXT,
  "pendingUpload" BOOLEAN DEFAULT false,
  "localAttachmentData" TEXT,
  created_at TIMESTAMPTZ DEFAULT now(),
  updated_at TIMESTAMPTZ DEFAULT now(),
  device_id TEXT
);

-- Quick Upgrade for existing Supabase projects:
-- ALTER TABLE temporary_slips ADD COLUMN IF NOT EXISTS "tokenNumber" TEXT;
-- ALTER TABLE bills ADD COLUMN IF NOT EXISTS "tokenNumber" TEXT;

CREATE TABLE IF NOT EXISTS transfers (
  id BIGSERIAL PRIMARY KEY,
  date TEXT,
  type TEXT,
  amount NUMERIC DEFAULT 0,
  remarks TEXT,
  created_at TIMESTAMPTZ DEFAULT now(),
  updated_at TIMESTAMPTZ DEFAULT now(),
  device_id TEXT
);

CREATE TABLE IF NOT EXISTS vendors (
  id BIGSERIAL PRIMARY KEY,
  name TEXT UNIQUE NOT NULL,
  category TEXT DEFAULT 'General',
  phone TEXT,
  remarks TEXT,
  created_at TIMESTAMPTZ DEFAULT now(),
  updated_at TIMESTAMPTZ DEFAULT now(),
  device_id TEXT
);

CREATE TABLE IF NOT EXISTS heads (
  id BIGSERIAL PRIMARY KEY,
  name TEXT UNIQUE NOT NULL,
  created_at TIMESTAMPTZ DEFAULT now(),
  updated_at TIMESTAMPTZ DEFAULT now(),
  device_id TEXT
);

CREATE TABLE IF NOT EXISTS upi_reconciliations (
  id BIGSERIAL PRIMARY KEY,
  date TEXT NOT NULL,
  hospital_upi NUMERIC DEFAULT 0,
  bank_upi NUMERIC DEFAULT 0,
  difference NUMERIC DEFAULT 0,
  status TEXT DEFAULT 'matched',
  remarks TEXT,
  created_at TIMESTAMPTZ DEFAULT now(),
  updated_at TIMESTAMPTZ DEFAULT now(),
  device_id TEXT
);

-- Quick Upgrade for existing Supabase projects (run once):
-- ALTER TABLE bills ADD COLUMN IF NOT EXISTS "head" TEXT;

-- Enable Row Level Security (optional but recommended)
ALTER TABLE settings ENABLE ROW LEVEL SECURITY;
ALTER TABLE advance_cash ENABLE ROW LEVEL SECURITY;
ALTER TABLE hospital_cash ENABLE ROW LEVEL SECURITY;
ALTER TABLE temporary_slips ENABLE ROW LEVEL SECURITY;
ALTER TABLE bills ENABLE ROW LEVEL SECURITY;
ALTER TABLE transfers ENABLE ROW LEVEL SECURITY;
ALTER TABLE hospital_deposits ENABLE ROW LEVEL SECURITY;
ALTER TABLE vendors ENABLE ROW LEVEL SECURITY;
ALTER TABLE heads ENABLE ROW LEVEL SECURITY;
ALTER TABLE upi_reconciliations ENABLE ROW LEVEL SECURITY;
